# Launch the gallery from the launcher, run its self-test, use some widgets,
# then return with the Home (Windows) key.
expect Keyboard attached 60
expect launcher ready 60
wait 1
shot 01-launcher
# tiles: dice39, gallery, hello (test builds: then qrscan, totp, crashtest, qjstests)
key right
key ret
expect Starting gallery
expect selftest: \d+ passed, 0 failed 30
wait 1
shot 02-gallery
# focus is on the letters-only text area: digits must be rejected
type abc123def
expect text: abcdef$
# held keys auto-repeat while typing (Enter and Space on buttons do not)
key a@1500
expect text: abcdefaaa+$
key tab
key spc
expect checkbox: true
key tab
key spc
expect switch: true
shot 03-inputs
# the pointer area under the slider: a click, then a drag
moveto 100 290
click
expect pointer: press \d+,\d+
expect pointer: release after 0 drag
mousedown
mouse 60 0
mouse 0 20
mouseup
expect pointer: release after [1-9]\d* drag
# self-test tab: click its header (5th tab of 5)
moveto 90% 20
click
wait 1
shot 04-selftest
# Home key back to the launcher
key meta_l
expect Stopping gallery
expect launcher ready
wait 1
shot 05-back
