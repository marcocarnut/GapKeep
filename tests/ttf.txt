# Needs a test build (make TEST=1): bold and italic (src/runtime/ttfstyles.h),
# what they cost. Boot makes none ("fonts:" only once a label uses one); a
# bold paragraph's first draw rasterizes its glyphs, a redraw takes them
# from the cache ("#fonts", "#redraw": src/testshot.h); tests/apps/ttftest
# logs LVGL's memory after each. The numbers are for reading, not checked.
expect Keyboard attached 60
expect launcher ready 60
tile ttftest
expect ttftest ready 30
mark
send #redraw
expect redraw: \d+ us 10
# Medium (bitmaps), then bold: first draw, then from the cache
key m
expect ttftest m 16: ui 10
send #redraw
expect redraw: \d+ us 10
mark
key b
expect fonts: bold 16 px: made 10
expect ttftest b 16: ui 10
send #fonts
expect fonts: bold 16 px: \d+ glyph bitmaps 10
mark
send #redraw
expect redraw: \d+ us 10
send #fonts
expect fonts: bold 16 px: \d+ glyph bitmaps 10
mark
send #redraw
expect redraw: \d+ us 10
shot 01-bold-16
# italic, and bold at 28 px
key i
expect ttftest i 16: ui 10
send #fonts
expect fonts: italic 16 px: \d+ glyph bitmaps 10
mark
key shift-b
expect ttftest b 28: ui 10
send #fonts
expect fonts: bold 28 px: \d+ glyph bitmaps 10
mark
send #redraw
expect redraw: \d+ us 10
shot 02-bold-28
